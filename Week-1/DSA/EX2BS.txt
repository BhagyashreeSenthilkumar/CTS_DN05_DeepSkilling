public class EX2BS {
    public static void main(String[] args) {

        int[] products = {101, 102, 103, 104, 105};
        int key = 104;

        int low = 0;
        int high = products.length - 1;

        while (low <= high) {

            int mid = (low + high) / 2;

            if (products[mid] == key) {
                System.out.println("Found at index " + mid);
                return;
            }

            if (products[mid] < key)
                low = mid + 1;
            else
                high = mid - 1;
        }

        System.out.println("Not Found");
    }
}
