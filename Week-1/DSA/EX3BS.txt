public class EX3BS {

    public static void main(String[] args) {

        int[] orders = {5000, 2000, 8000, 1000};

        for (int i = 0; i < orders.length - 1; i++) {
            for (int j = 0; j < orders.length - i - 1; j++) {

                if (orders[j] > orders[j + 1]) {
                    int temp = orders[j];
                    orders[j] = orders[j + 1];
                    orders[j + 1] = temp;
                }
            }
        }

        for (int order : orders)
            System.out.print(order + " ");
    }
}
